cmake_minimum_required(VERSION 3.16)
project(native_ipc LANGUAGES CXX)
add_library(native_ipc STATIC
 project/src/Common.cpp project/src/PipeWindows.cpp project/src/PipePosix.cpp
 project/src/Channel.cpp project/src/SharedMemory.cpp)
target_include_directories(native_ipc PUBLIC project/include)
target_compile_features(native_ipc PUBLIC cxx_std_11)
find_package(Threads REQUIRED)
target_link_libraries(native_ipc PUBLIC Threads::Threads)
if(WIN32)
 target_link_libraries(native_ipc PUBLIC advapi32)
elseif(UNIX AND NOT APPLE)
 target_link_libraries(native_ipc PUBLIC rt)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(native_ipc_tests tests/NativeTests.cpp)
 target_link_libraries(native_ipc_tests PRIVATE native_ipc)
 add_test(NAME native_ipc COMMAND native_ipc_tests)
 set_tests_properties(native_ipc PROPERTIES TIMEOUT 60)
endif()
